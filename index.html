function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var query = e.parameter.q ? e.parameter.q.toString().trim().toLowerCase() : ""; var result = { status: "not_found" }; // 1-жол тақырып деп есептейміз, 2-жолдан бастап іздейміз for (var i = 1; i < rows.length; i++) { var row = rows[i]; var fullName = row[0] ? row[0].toString().trim() : ""; // 1-ші баған: Аты-жөні var regCode = row[1] ? row[1].toString().trim() : ""; // 2-ші баған: Тіркеу коды if (fullName.toLowerCase().includes(query) || regCode.toLowerCase().includes(query)) { result = { status: "success", owner: fullName, code: regCode, docType: row[2] ? row[2].toString() : "Сертификат / Куәлік", org: row[3] ? row[3].toString() : "«BILIM QAZYNA» КММ", date: row[4] ? row[4].toString() : "17.09.2026" }; break; } } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
